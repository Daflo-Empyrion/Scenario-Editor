# Ressource de version Windows pour l'executable -- un .exe sans metadonnees
# (editeur, description, version) ressemble davantage a un malware aux yeux des
# heuristiques antivirus qu'un executable qui en a. Voir empyrion_editor.spec
# (parametre version=... des blocs EXE) et BUILD.md pour la procedure de mise a
# jour a chaque nouvelle version.
#
# A CHAQUE NOUVELLE VERSION : mets a jour filevers/prodvers ci-dessous pour
# qu'ils correspondent a APP_VERSION (core/version.py) et MyAppVersion
# (installer.iss) -- format a 4 chiffres, donc "1.0.10" devient (1, 0, 10, 0).

VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 3, 4, 0),
    prodvers=(1, 3, 4, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo(
      [
      StringTable(
        u'040904b0',
        [StringStruct(u'CompanyName', u'Daflo'),
        StringStruct(u'FileDescription', u'Empyrion Scenario Editor -- editeur de scenario pour Empyrion Galactic Survival'),
        StringStruct(u'FileVersion', u'1.3.4.0'),
        StringStruct(u'InternalName', u'EmpyrionScenarioEditor'),
        StringStruct(u'LegalCopyright', u'Copyright (C) 2026 Daflo -- GNU GPLv3'),
        StringStruct(u'OriginalFilename', u'EmpyrionScenarioEditor.exe'),
        StringStruct(u'ProductName', u'Empyrion Scenario Editor'),
        StringStruct(u'ProductVersion', u'1.3.4.0')])
      ]),
    VarFileInfo([VarStruct(u'Translation', [1036, 1200])])
  ]
)
